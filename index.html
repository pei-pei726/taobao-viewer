    <script>

        let allData = [];


        window.addEventListener('DOMContentLoaded', function() {

            const savedData = localStorage.getItem('taobao_orders');

            if (savedData) {

                try {

                    allData = JSON.parse(savedData);

                    if (allData.length > 0) {

                        processData(allData);

                    }

                } catch (e) {

                    console.error('解析儲存的資料失敗', e);

                }

            }

        });


        document.getElementById('fileInput').addEventListener('change', function(e) {

            const file = e.target.files[0];

            if (!file) return;


            const reader = new FileReader();

            reader.onload = function(e) {

                const data = new Uint8Array(e.target.result);

                const workbook = XLSX.read(data, {type: 'array'});

                const firstSheetName = workbook.SheetNames[0];

                const worksheet = workbook.Sheets[firstSheetName];

                

                const newRows = XLSX.utils.sheet_to_json(worksheet);

                mergeNewData(newRows);

            };

            reader.readAsArrayBuffer(file);

            this.value = '';

        });


        function mergeNewData(newRows) {

            let addedCount = 0;

            const getOrderId = (row) => String(row['訂單編號'] || row['主訂單編號'] || row['子訂單編號'] || '');

            const existingIds = new Set(allData.map(row => getOrderId(row)).filter(id => id !== ''));


            newRows.forEach(row => {

                const orderId = getOrderId(row);

                if (!orderId || !existingIds.has(orderId)) {

                    allData.push(row);

                    if (orderId) existingIds.add(orderId);

                    addedCount++;

                }

            });


            localStorage.setItem('taobao_orders', JSON.stringify(allData));

            alert(`成功匯入！本次新增了 ${addedCount} 筆新訂單。`);

            processData(allData);

        }


        document.getElementById('clearBtn').addEventListener('click', function() {

            if (confirm('確定要清除所有已儲存的歷史資料嗎？')) {

                localStorage.removeItem('taobao_orders');

                allData = [];

                document.getElementById('dashboard').classList.add('hidden');

                alert('已清除所有資料！');

            }

        });


        function processData(data) {

            if (data.length === 0) return;

            document.getElementById('dashboard').classList.remove('hidden');

            renderTable(data);

        }


        function renderTable(data) {

            const tbody = document.getElementById('tableBody');

            tbody.innerHTML = '';


            let totalSum = 0;

            document.getElementById('totalCount').innerText = `${data.length} 筆`;


            data.forEach(row => {

                const title = row['商品標題'] || row['寶貝標題'] || row['商品名稱'] || '未知商品';

                const price = parseFloat(row['實付款'] || row['單價'] || row['價格'] || 0);

                const date = row['成交時間'] || row['購買時間'] || row['訂單創建時間'] || '-';

                const imgUrl = row['寶貝圖片'] || row['商品圖片'] || row['圖片'] || '';

                const itemUrl = row['寶貝鏈接'] || row['商品鏈接'] || row['連結'] || '#';


                totalSum += isNaN(price) ? 0 : price;


                const tr = document.createElement('tr');

                tr.className = 'hover:bg-gray-50 transition-colors';

                

                tr.innerHTML = `

                    <td class="p-4 w-20">

                        ${imgUrl ? `<img src="${imgUrl}" class="w-16 h-16 object-cover rounded-lg border border-gray-200">` : '<div class="w-16 h-16 bg-gray-100 rounded-lg flex items-center justify-center text-xs text-gray-400">無圖</div>'}

                    </td>

                    <td class="p-4">

                        <a href="${itemUrl}" target="_blank" class="font-medium text-gray-800 hover:text-orange-600 line-clamp-2">${title}</a>

                    </td>

                    <td class="p-4 font-semibold text-gray-700 whitespace-nowrap">¥ ${price.toFixed(2)}</td>

                    <td class="p-4 text-gray-500 whitespace-nowrap">${date}</td>

                `;

                tbody.appendChild(tr);

            });


            document.getElementById('totalAmount').innerText = `¥ ${totalSum.toFixed(2)}`;

        }


        document.getElementById('searchInput').addEventListener('input', function(e) {

            const keyword = e.target.value.toLowerCase();

            const filtered = allData.filter(row => {

                const title = String(row['商品標題'] || row['寶貝標題'] || row['商品名稱'] || '').toLowerCase();

                return title.includes(keyword);

            });

            renderTable(filtered);

        });
